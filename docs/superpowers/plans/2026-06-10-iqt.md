```ts
export type Category = 'patterns' | 'sequences' | 'spatial'
export type Difficulty = 'easy' | 'medium' | 'hard'
export type Shape =
  | 'circle' | 'square' | 'triangle' | 'diamond' | 'pentagon' | 'hexagon'
  | 'star' | 'cross' | 'arrow' | 'dot' | 'L' | 'flag'
export type Fill = 'solid' | 'outline' | 'hatch' | 'half-left' | 'half-right' | 'half-top' | 'half-bottom'

export interface Glyph {
  shape: Shape
  fill?: Fill        // default 'solid'
  rotation?: number  // degrees, clockwise, around cell center
  scale?: number     // default 1
  count?: number     // only for shape 'dot': 1–9 dice-style dots
  mirror?: boolean   // horizontal flip (for chiral shapes L/flag)
}
export type Cell = Glyph[] // layered glyphs, drawn in order

interface PuzzleBase {
  id: string
  category: Category
  difficulty: Difficulty
  options: Cell[]    // 4–6 answer cells
  correct: number    // index into options
}
export interface MatrixPuzzle extends PuzzleBase { kind: 'matrix'; grid: Cell[] }       // 8 cells row-major, bottom-right missing
export interface SequencePuzzle extends PuzzleBase { kind: 'sequence'; sequence: Cell[] } // 3–4 cells, next is missing
export interface OddOneOutPuzzle extends PuzzleBase { kind: 'odd-one-out' }              // options ARE the items; correct = the odd one
export interface RotationPuzzle extends PuzzleBase { kind: 'rotation'; base: Cell }      // pick rotation of base (distractors mirrored)
export type Puzzle = MatrixPuzzle | SequencePuzzle | OddOneOutPuzzle | RotationPuzzle
```

```ts
import type { Category, Difficulty } from './types'

export const QUESTION_SECONDS = 35
const POINTS: Record<Difficulty, number> = { easy: 1, medium: 2, hard: 3 }

export interface RunRecord {
  correct: boolean
  category: Category
  difficulty: Difficulty
  remaining: number // seconds left when answered
}

export function rawScore(records: { correct: boolean; difficulty: Difficulty; remaining: number }[]): number {
  return records.reduce((sum, r) => {
    if (!r.correct) return sum
    const bonus = 0.5 * Math.max(0, Math.min(1, r.remaining / QUESTION_SECONDS))
    return sum + POINTS[r.difficulty] + bonus
  }, 0)
}

export function maxRaw(records: { difficulty: Difficulty }[]): number {
  return records.reduce((sum, r) => sum + POINTS[r.difficulty] + 0.5, 0)
}

export function toIQ(raw: number, max: number): number {
  const iq = 70 + 75 * (raw / max)
  return Math.round(Math.min(145, Math.max(70, iq)))
}

// Standard normal CDF via Abramowitz–Stegun erf approximation (|error| < 1.5e-7)
export function percentile(iq: number): number {
  const z = (iq - 100) / 15
  const x = Math.abs(z) / Math.SQRT2
  const t = 1 / (1 + 0.3275911 * x)
  const poly = ((((1.061405429 * t - 1.453152027) * t + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t
  const erf = 1 - poly * Math.exp(-x * x)
  const cdf = 0.5 * (1 + Math.sign(z) * erf)
  return Math.round(cdf * 100)
}

export interface Title { name: string; emoji: string }
export function titleFor(iq: number): Title {
  if (iq >= 130) return { name: 'Pattern Prodigy', emoji: '🧬' }
  if (iq >= 115) return { name: 'Logic Llama', emoji: '🦙' }
  if (iq >= 100) return { name: 'Puzzle Pathfinder', emoji: '🧭' }
  if (iq >= 85) return { name: 'Curious Cadet', emoji: '🚀' }
  return { name: 'Rogue Variable', emoji: '🎲' }
}
```

```ts
import type { Puzzle, Category, Difficulty, Cell } from './types'

export type RNG = () => number
export interface RunQuestion { puzzle: Puzzle; options: Cell[]; correct: number }

const CATEGORIES: Category[] = ['patterns', 'sequences', 'spatial']
const QUOTA: [Difficulty, number][] = [['easy', 2], ['medium', 2], ['hard', 1]]
const DIFF_ORDER: Difficulty[] = ['easy', 'medium', 'hard']

function shuffle<T>(items: T[], rng: RNG): T[] {
  const a = [...items]
  for (let i = a.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1))
    ;[a[i], a[j]] = [a[j], a[i]]
  }
  return a
}

export function selectRun(pool: Puzzle[], rng: RNG): RunQuestion[] {
  const picked: Puzzle[] = []
  for (const category of CATEGORIES) {
    for (const [difficulty, n] of QUOTA) {
      const candidates = pool.filter((p) => p.category === category && p.difficulty === difficulty)
      picked.push(...shuffle(candidates, rng).slice(0, n))
    }
  }
  const ordered = DIFF_ORDER.flatMap((d) => shuffle(picked.filter((p) => p.difficulty === d), rng))
  return ordered.map((puzzle) => {
    const perm = shuffle(puzzle.options.map((_, i) => i), rng)
    return { puzzle, options: perm.map((i) => puzzle.options[i]), correct: perm.indexOf(puzzle.correct) }
  })
}
```

```svelte
<script lang="ts">
  import type { Cell, Glyph } from '../lib/types'
  import { SHAPE_PATHS, DOT_LAYOUTS } from '../lib/shapes'

  let { cell, size = 72 }: { cell: Cell; size?: number } = $props()
  const uid = Math.random().toString(36).slice(2, 8)
  const INK = '#7dd3fc'

  function transform(g: Glyph): string {
    const s = g.scale ?? 1
    return `translate(50 50) rotate(${g.rotation ?? 0}) scale(${g.mirror ? -s : s} ${s}) translate(-50 -50)`
  }
  type Layer = { fill: string; stroke: boolean; clip: string | null }
  function layers(g: Glyph): Layer[] {
    const f = g.fill ?? 'solid'
    if (f === 'solid') return [{ fill: INK, stroke: false, clip: null }]
    if (f === 'outline') return [{ fill: 'none', stroke: true, clip: null }]
    if (f === 'hatch') return [{ fill: `url(#hatch-${uid})`, stroke: true, clip: null }]
    return [
      { fill: 'none', stroke: true, clip: null },
      { fill: INK, stroke: false, clip: `url(#${f.slice(5)}-${uid})` },
    ]
  }
</script>
```

```bash
#!/usr/bin/env bash
set -euo pipefail
cd "$(dirname "$0")"
HOST=root@143.110.208.106 # iqt.moosha.org

npm test
npm run build

ssh "$HOST" 'mkdir -p /opt/iqt'
rsync -az --delete dist/ "$HOST":/opt/iqt/dist/
rsync -az Dockerfile docker-compose.yml "$HOST":/opt/iqt/
ssh "$HOST" 'cd /opt/iqt && docker compose up -d --build'

# one-time Caddy vhost
ssh "$HOST" 'grep -q "iqt.moosha.org" /etc/caddy/Caddyfile || {
  cat >> /etc/caddy/Caddyfile <<'"'"'EOF'"'"'

iqt.moosha.org {
	reverse_proxy localhost:8015
	encode gzip zstd
	header {
		X-Content-Type-Options nosniff
		X-Frame-Options DENY
		Referrer-Policy strict-origin-when-cross-origin
		Strict-Transport-Security "max-age=31536000"
		-Server
	}
	@static path /assets/*
	header @static Cache-Control "public, max-age=604800, immutable"
}
EOF
  systemctl reload caddy
}'
echo "✅ https://iqt.moosha.org"
```
